function move(g,i,to,pos){
  if(!DOC[to])return;
  var T=DOC[to].items,it=DOC[g].items[i],f=-1,k;
  if(pos===null&&g===to)return;
  DOC[g].items.splice(i,1);
  if(g!==to)for(k=0;k<T.length;k++)if(T[k].n===it.n&&T[k].p===it.p)f=k;
  if(f>=0)T[f].q+=it.q;
  else{if(pos===null)pos=T.length;if(g===to&&i<pos)pos--;T.splice(pos,0,it);}
  render();
}
